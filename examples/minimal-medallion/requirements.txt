# Local dev dependencies for the minimal-medallion example.
#
# Versions are pinned to Microsoft Fabric Runtime 2.0 (Spark 4.1 / Delta 4.2)
# so local execution matches what runs in Fabric. pyfabric-dev itself only
# floors these (pyspark>=4.1.1, delta-spark>=4.2.0); without the pins below, pip
# would resolve to the latest Spark/Delta, which no Fabric runtime uses — so
# always install via this file, not a bare `pip install pyfabric-dev`.

pyfabric-dev>=0.6.0

# Spark + Delta — match Fabric Runtime 2.0.
pyspark==4.1.1
delta-spark==4.2.0

# Data processing — major versions match Fabric Runtime 2.0 (pandas 2.3.3, numpy 2.4.1).
pandas>=2.3.3,<3.0
numpy>=2.4,<3.0

# Testing
pytest>=7.0.0
